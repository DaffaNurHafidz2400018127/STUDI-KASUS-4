  #include <iostream>
  #include <vector>
  using namespace std;
  
  class Bulanan {
  public:
      int bulan;
      int pengeluaran;
      int tabungan;
  
        Bulanan(int b, int p, int t) {
            bulan = b;
            pengeluaran = p;
            tabungan = t;
        }
  };
   
  class PengelolaKeuangan {
  private:
      vector<Bulanan> dataBulanan;
      int uangSaku;
  
  public:
      PengelolaKeuangan(int uang) {
          uangSaku = uang;
      }
  
      void tambahData(int pengeluaran, int tabungan) {
          int bulanKe = dataBulanan.size() + 1;
          dataBulanan.push_back(Bulanan(bulanKe, pengeluaran, tabungan));
      }
  
      void tampilkanDataIteratif() {
          cout << "\nData Keuangan Tiap Bulan:\n";
          for (int i = 0; i < dataBulanan.size(); i++) {
              cout << "Bulan " << dataBulanan[i].bulan
                   << ": Pengeluaran = " << dataBulanan[i].pengeluaran
                   << ", Tabungan = " << dataBulanan[i].tabungan << endl;
          }
      }
  
      int hitungTotalPengeluaran(int index = 0) {
          if (index >= dataBulanan.size()) return 0;
          return dataBulanan[index].pengeluaran + hitungTotalPengeluaran(index + 1);
      }
  
      int hitungTotalTabungan(int index = 0) {
          if (index >= dataBulanan.size()) return 0;
          return dataBulanan[index].tabungan + hitungTotalTabungan(index + 1);
      }
  };
  
  class Laporan {
  public:
      static void tampilkanLaporan(PengelolaKeuangan &pengelola) {
          pengelola.tampilkanDataIteratif();
          cout << "\nTotal Pengeluaran: " << pengelola.hitungTotalPengeluaran();
          cout << "\nTotal Tabungan: " << pengelola.hitungTotalTabungan() << endl;
      }
  };
  
  int main() {
      int uangSaku, bulan, pengeluaran, tabungan;
  
      cout << "Masukkan jumlah uang saku Andi per bulan: ";
      cin >> uangSaku;
  
      cout << "Masukkan jumlah bulan: ";
      cin >> bulan;
  
      PengelolaKeuangan pengelola(uangSaku);
  
      for (int i = 0; i < bulan; i++) {
          cout << "\nBulan " << i + 1 << ":\n";
          cout << "Pengeluaran: ";
          cin >> pengeluaran;
          cout << "Tabungan: ";
          cin >> tabungan;
  
          pengelola.tambahData(pengeluaran, tabungan);
      }
  
      Laporan::tampilkanLaporan(pengelola);
  
      return 0;
  }
